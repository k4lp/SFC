  ```csharp
  private async Task<List<string>> GetPermissionedFieldsAsync<T>(
      CancellationToken cancellationToken) where T : class, new()
  {
      var mappedFields = SalesforceMapper.GetQueryableFields<T>().ToList();
      
      if (_flsService == null)
          return mappedFields;
      
      var objectName = SalesforceMapper.GetObjectName<T>();
      var readableFields = await _flsService.GetReadableFieldsAsync(objectName, cancellationToken);
      var readableNames = new HashSet<string>(
          readableFields.Select(f => f.Name), 
          StringComparer.OrdinalIgnoreCase);
      
      return mappedFields.Where(f => 
          f.Equals("Id", StringComparison.OrdinalIgnoreCase) || 
          readableNames.Contains(f)).ToList();
  }
  ```

PHASE 5: SOQLBUILDER ENHANCEMENTS (SoqlBuilder.cs)

Step 5.1: Add WithFlsEnforcement Method
  File: src/SalesforceCore/Services/Query/SoqlBuilder.cs
  Location: After Build() method (around line 1400)
  
  Add:
  ```csharp
  /// <summary>
  /// Validates and optionally filters selected fields based on FLS.
  /// </summary>
  /// <param name="flsService">The FLS service to use for validation.</param>
  /// <param name="mode">How to handle inaccessible fields.</param>
  /// <param name="cancellationToken">Cancellation token.</param>
  /// <returns>This builder for chaining.</returns>
  public async Task<SoqlBuilder> WithFlsEnforcementAsync(
      IFieldLevelSecurityService flsService,
      FlsEnforcementMode mode,
      CancellationToken cancellationToken = default)
  {
      if (mode == FlsEnforcementMode.None)
          return this;
      
      var readableFields = await flsService.GetReadableFieldsAsync(_objectName, cancellationToken);
      var readableNames = new HashSet<string>(
          readableFields.Select(f => f.Name), 
          StringComparer.OrdinalIgnoreCase);
      
      var inaccessible = _selectFields
          .Where(f => !readableNames.Contains(f) && !IsAggregateOrFunction(f))
          .ToList();
      
      if (inaccessible.Count == 0)
          return this;
      
      if (mode == FlsEnforcementMode.Strict)
      {
          throw new FlsException(inaccessible.Select(f => new FlsViolation(
              f, f, FlsOperation.Read, 
              $"Field '{f}' is not readable on {_objectName}.")));
      }
      
      // Silent mode: remove inaccessible fields
      _selectFields.RemoveAll(f => inaccessible.Contains(f));
      return this;
  }
  
  private static bool IsAggregateOrFunction(string field)
  {
      return field.Contains("(") || field.StartsWith("COUNT", StringComparison.OrdinalIgnoreCase);
  }
  ```

  File: src/SalesforceCore/Security/Explanation.md
  
  Add new section after "Design Decisions":
  ```markdown
  ## 4. FLS Enforcement Modes
  
  The `FlsEnforcementMode` setting in `SalesforceOptions` controls behavior:
  
  | Mode   | Read Operations          | Write Operations         |
  |--------|--------------------------|--------------------------|
  | Silent | Fields quietly excluded  | Fields quietly dropped   |
  | Strict | FlsException thrown      | FlsException thrown      |
  | None   | No checks performed      | No checks performed      |
  
  **Recommendation**: Use `Strict` mode in development to catch permission issues early.
  Use `Silent` mode in production for graceful degradation.
  ```

Step 6.2: Update Services/Data/Explanation.md
  File: src/SalesforceCore/Services/Data/Explanation.md
  Location: Line 11 (Automatic FLS Checks bullet)
  
  Update to:
  ```markdown
  - **Automatic FLS Checks**: Before sending data, it consults `ISchemaService` to filter out fields 
    that the user doesn't have permission to create or update. Behavior depends on `FlsEnforcementMode`:
    - `Silent`: Inaccessible fields are quietly removed (default, backward compatible).
    - `Strict`: Throws `FlsException` listing all inaccessible fields.
    - `None`: No filtering; Salesforce API will return errors for inaccessible fields.
  ```
